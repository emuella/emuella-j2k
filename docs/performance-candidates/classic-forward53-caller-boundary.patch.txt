+    width: usize,

+    workers: usize,
+    coefficients: &mut [Vec<i32>],
+) -> Result<bool> {

+    #[cfg(feature = "classic-execution-diagnostics")]
+    let preparation = EncodeClock::start::<PROFILE>();
+    let prepared = prepare_forward53(width, height, workers)?;
+    #[cfg(feature = "classic-execution-diagnostics")]
+    if PROFILE {

+    let dropping = EncodeClock::start::<PROFILE>();

+    #[cfg(feature = "classic-execution-diagnostics")]
+    if PROFILE {
+        diagnostics::update(|d| d.dwt_scratch_drop_ns += dropping.ns());
